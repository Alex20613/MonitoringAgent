cmake_minimum_required(VERSION 3.10)

project(AgenteRemoto LANGUAGES CXX)

# Habilita C++17
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Buscar mosquitto (instalado con vcpkg)
find_package(unofficial-mosquitto CONFIG REQUIRED)

# Agrega tu ejecutable
add_executable(agente agent.cpp)

# Incluye las rutas necesarias
target_include_directories(agente PRIVATE ${CMAKE_SOURCE_DIR}/include)

# Enlazar mosquitto
target_link_libraries(agente PRIVATE unofficial::mosquitto::mosquitto)
